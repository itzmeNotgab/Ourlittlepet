name: Deploy Firebase (rules + functions)

on:
  workflow_dispatch:
    inputs:
      project_id:
        description: Firebase project ID
        required: true

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Install tools
        run: |
          npm install -g firebase-tools
          npm --prefix functions install --no-audit --no-fund
      - name: Write service account (from secret, never committed)
        run: echo '${{ secrets.FIREBASE_SERVICE_ACCOUNT }}' > "$RUNNER_TEMP/sa.json"
      - name: Deploy
        env:
          GOOGLE_APPLICATION_CREDENTIALS: ${{ runner.temp }}/sa.json
        run: firebase deploy --project "${{ github.event.inputs.project_id }}" --only firestore,storage,database,functions --non-interactive
